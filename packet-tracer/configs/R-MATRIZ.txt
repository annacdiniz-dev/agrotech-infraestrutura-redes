! =====================================================================
!  R-MATRIZ  - Cisco 2911 - Roteador da matriz (Goiania)
!  Funcao: router-on-a-stick (roteamento entre VLANs), relay DHCP,
!          ACL da rede de visitantes e saida para o provedor (R-ISP).
!  Ligacoes: Gi0/0 -> SW-CORE Gi0/1 (trunk 802.1Q)
!            Gi0/1 -> R-ISP Gi0/0   (WAN 203.0.113.0/30)
!  Como usar: CLI do roteador -> responda "no" ao "initial configuration
!  dialog" -> cole este arquivo inteiro (ou bloco a bloco).
! =====================================================================
enable
configure terminal
!
! ---------- Identificacao e seguranca basica ----------
hostname R-MATRIZ
no ip domain-lookup
enable secret AgroTech2026
service password-encryption
banner motd #Acesso restrito - AgroTech Solucoes Agricolas - R-MATRIZ#
line vty 0 4
 password AgroTech2026
 login
 exit
line console 0
 logging synchronous
 exit
!
! ---------- ACL da VLAN 50 (Wi-Fi visitantes) ----------
! Visitante so pode: pedir IP (DHCP), consultar o DNS interno e sair
! para a Internet. Qualquer acesso as redes internas 192.168.0.0/16 e negado.
ip access-list extended VISITANTES-MATRIZ
 remark DHCP (DISCOVER/REQUEST do visitante, repassado pelo ip helper)
 permit udp any any eq bootps
 remark Consulta ao DNS interno (somente porta 53)
 permit udp 192.168.50.0 0.0.0.255 host 192.168.30.11 eq domain
 remark Bloqueia todas as redes internas da AgroTech
 deny ip 192.168.50.0 0.0.0.255 192.168.0.0 0.0.255.255
 remark Libera o restante (Internet)
 permit ip 192.168.50.0 0.0.0.255 any
 exit
!
! ---------- Interface fisica do trunk (sem IP) ----------
interface GigabitEthernet0/0
 description TRUNK 802.1Q para SW-CORE Gi0/1
 no ip address
 no shutdown
 exit
!
! ---------- Subinterfaces: uma por VLAN (gateway .1 de cada rede) ----------
interface GigabitEthernet0/0.10
 description VLAN 10 - Administrativo/Financeiro
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip helper-address 192.168.30.10
 exit
interface GigabitEthernet0/0.20
 description VLAN 20 - Comercial/RH/Recepcao
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip helper-address 192.168.30.10
 exit
interface GigabitEthernet0/0.30
 description VLAN 30 - TI/Servidores (DHCP na propria VLAN, sem helper)
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 exit
interface GigabitEthernet0/0.40
 description VLAN 40 - Cameras IP/IoT
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip helper-address 192.168.30.10
 exit
interface GigabitEthernet0/0.50
 description VLAN 50 - Wi-Fi Visitantes (somente Internet)
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0
 ip helper-address 192.168.30.10
 ip access-group VISITANTES-MATRIZ in
 exit
interface GigabitEthernet0/0.60
 description VLAN 60 - VoIP
 encapsulation dot1Q 60
 ip address 192.168.60.1 255.255.255.0
 ip helper-address 192.168.30.10
 exit
interface GigabitEthernet0/0.99
 description VLAN 99 - Nativa do trunk (sem uso para dados)
 encapsulation dot1Q 99 native
 exit
!
! ---------- WAN: link com o provedor ----------
interface GigabitEthernet0/1
 description WAN - Provedor (R-ISP Gi0/0) - 203.0.113.0/30
 ip address 203.0.113.1 255.255.255.252
 no shutdown
 exit
!
! ---------- Rotas estaticas ----------
! Redes da filial (Anapolis), alcancadas atraves do provedor.
! Em producao, este caminho seria o tunel VPN IPSec do pfSense.
ip route 192.168.110.0 255.255.255.0 203.0.113.2
ip route 192.168.120.0 255.255.255.0 203.0.113.2
ip route 192.168.150.0 255.255.255.0 203.0.113.2
! Rota padrao: todo o resto (Internet) vai para o provedor.
ip route 0.0.0.0 0.0.0.0 203.0.113.2
!
end
write memory
