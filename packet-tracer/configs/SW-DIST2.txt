! =====================================================================
!  SW-DIST2  - Cisco 2960-24TT - Distribuicao 2 (TI, servidores e cameras)
!  (No projeto real: TP-Link TL-SG3428.)
!  Ligacoes: Gi0/1  -> SW-CORE Fa0/24 (trunk)
!            Fa0/1  -> SRV-DHCP (192.168.30.10)   VLAN 30
!            Fa0/2  -> SRV-DNS  (192.168.30.11)   VLAN 30
!            Fa0/3  -> SRV-WEB  (192.168.30.12)   VLAN 30
!            Fa0/4  -> SRV-FTP  (192.168.30.13)   VLAN 30
!            Fa0/5  -> PC-TI                      VLAN 30
!            Fa0/13 -> PC-CAM (simula camera IP)  VLAN 40
! =====================================================================
enable
configure terminal
!
! ---------- Identificacao ----------
hostname SW-DIST2
no ip domain-lookup
enable secret AgroTech2026
service password-encryption
banner motd #Acesso restrito - AgroTech - SW-DIST2#
line console 0
 logging synchronous
 exit
!
vtp mode transparent
!
! ---------- VLANs ----------
vlan 10
 name ADM-FINANCEIRO
vlan 20
 name COMERCIAL-RH
vlan 30
 name TI-SERVIDORES
vlan 40
 name CAMERAS-IOT
vlan 50
 name WIFI-VISITANTES
vlan 60
 name VOIP
vlan 99
 name NATIVA
vlan 999
 name PORTAS-NAO-USADAS
 exit
!
spanning-tree mode rapid-pvst
!
! ---------- Uplink (trunk) ----------
interface GigabitEthernet0/1
 description TRUNK para SW-CORE Fa0/24
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60,99
 no shutdown
 exit
!
! ---------- Portas de acesso ----------
interface range FastEthernet0/1 - 12
 description VLAN 30 - TI/Servidores
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/13 - 20
 description VLAN 40 - Cameras IP/IoT
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/21 - 24 , GigabitEthernet0/2
 description Porta reservada (expansao)
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! ---------- Gerencia ----------
interface Vlan30
 description Gerencia do switch
 ip address 192.168.30.4 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.30.1
!
end
write memory
