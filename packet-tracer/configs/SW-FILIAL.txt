! =====================================================================
!  SW-FILIAL  - Cisco 2960-24TT - Switch da filial (Anapolis)
!  (No projeto real: TP-Link TL-SG3428 no rack 4U.)
!  Ligacoes: Gi0/1  -> R-FILIAL Gi0/0 (trunk)
!            Fa0/1  -> PC-FILIAL-ADM      (VLAN 110)
!            Fa0/9  -> PC-FILIAL-OFI      (VLAN 120)
!            Fa0/21 -> AP-FILIAL          (VLAN 150)
! =====================================================================
enable
configure terminal
!
! ---------- Identificacao ----------
hostname SW-FILIAL
no ip domain-lookup
enable secret AgroTech2026
service password-encryption
banner motd #Acesso restrito - AgroTech - SW-FILIAL#
line console 0
 logging synchronous
 exit
!
vtp mode transparent
!
! ---------- VLANs da filial ----------
vlan 110
 name ADM-ATENDIMENTO
vlan 120
 name OFICINA-ESTOQUE
vlan 150
 name WIFI-VISITANTES
vlan 99
 name NATIVA
vlan 999
 name PORTAS-NAO-USADAS
 exit
!
spanning-tree mode rapid-pvst
!
! ---------- Uplink para o roteador (trunk) ----------
interface GigabitEthernet0/1
 description TRUNK para R-FILIAL Gi0/0 (router-on-a-stick)
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 99,110,120,150
 no shutdown
 exit
!
! ---------- Portas de acesso ----------
interface range FastEthernet0/1 - 8
 description VLAN 110 - Administrativo/Atendimento
 switchport mode access
 switchport access vlan 110
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/9 - 16
 description VLAN 120 - Oficina/Estoque
 switchport mode access
 switchport access vlan 120
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/21 - 22
 description VLAN 150 - Access Point de visitantes
 switchport mode access
 switchport access vlan 150
 spanning-tree portfast
 no shutdown
 exit
interface range FastEthernet0/17 - 20 , FastEthernet0/23 - 24 , GigabitEthernet0/2
 description Porta reservada (expansao)
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! ---------- Gerencia (VLAN 110) ----------
interface Vlan110
 description Gerencia do switch
 ip address 192.168.110.2 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.110.1
!
end
write memory
