! =====================================================================
!  SW-DIST1  - Cisco 2960-24TT - Distribuicao 1 (bloco administrativo)
!  (No projeto real: TP-Link TL-SG3428.)
!  Atende: Administracao/Financeiro (VLAN 10), Comercial/RH/Recepcao
!          (VLAN 20), telefones IP (VLAN 60) e o AP de visitantes (VLAN 50)
!  Ligacoes: Fa0/1 -> SW-CORE Fa0/2 (trunk)
!            Fa0/2 -> PC-ADM            (VLAN 10)
!            Fa0/3 -> PC-COM            (VLAN 20)
!            Fa0/4 -> AP-VISITANTES     (VLAN 50)
!  Portas livres por VLAN (expansao): 10 = Fa0/5-10 | 20 = Fa0/11-16
!                                     60 = Fa0/17-20 | 50 = Fa0/21-22
! =====================================================================
enable
configure terminal
!
! ---------- Identificacao ----------
hostname SW-DIST1
no ip domain-lookup
enable secret AgroTech2026
service password-encryption
banner motd #Acesso restrito - AgroTech - SW-DIST1#
line console 0
 logging synchronous
 exit
!
vtp mode transparent
!
! ---------- VLANs ----------
vlan 10
 name ADM-FINANCEIRO
vlan 20
 name COMERCIAL-RH
vlan 30
 name TI-SERVIDORES
vlan 40
 name CAMERAS-IOT
vlan 50
 name WIFI-VISITANTES
vlan 60
 name VOIP
vlan 99
 name NATIVA
vlan 999
 name PORTAS-NAO-USADAS
 exit
!
spanning-tree mode rapid-pvst
!
! ---------- Uplink (trunk) ----------
interface FastEthernet0/1
 description TRUNK para SW-CORE Fa0/2
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60,99
 no shutdown
 exit
!
! ---------- Portas de acesso ----------
! PortFast: a porta entra direto em encaminhamento (sem esperar o STP),
! o que evita que o PC peca DHCP antes de a porta estar pronta.
! BPDU Guard: desliga a porta se alguem ligar um switch nela.
interface range FastEthernet0/2 , FastEthernet0/5 - 10
 description VLAN 10 - Administrativo/Financeiro
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/3 , FastEthernet0/11 - 16
 description VLAN 20 - Comercial/RH/Recepcao
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/17 - 20
 description VLAN 60 - Telefones IP (VoIP)
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface range FastEthernet0/4 , FastEthernet0/21 - 22
 description VLAN 50 - Access Point de visitantes
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
 no shutdown
 exit
interface range FastEthernet0/23 - 24 , GigabitEthernet0/1 - 2
 description Porta reservada (expansao)
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! ---------- Gerencia ----------
interface Vlan30
 description Gerencia do switch
 ip address 192.168.30.3 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.30.1
!
end
write memory
